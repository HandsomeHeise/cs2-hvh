cmake_minimum_required(VERSION 3.24)
project(hvh_menu LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "A Windows x64 toolchain is required")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/imgui/imgui.h")
    message(FATAL_ERROR "Run python tools/fetch_deps.py to fetch the pinned dependencies")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
set(IMGUI external/imgui)
set(MINHOOK external/minhook)
add_library(hvh_menu SHARED src/overlay.cpp src/menu.cpp src/config.cpp src/game_data.cpp src/skins.cpp src/aim.cpp src/server_rules.cpp src/esp.cpp
    ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp
    ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx11.cpp
    ${MINHOOK}/src/buffer.c ${MINHOOK}/src/hook.c ${MINHOOK}/src/trampoline.c ${MINHOOK}/src/hde/hde64.c)
target_include_directories(hvh_menu PRIVATE include src ${IMGUI} ${IMGUI}/backends ${MINHOOK}/include)
target_link_libraries(hvh_menu PRIVATE d3d11 dxgi d3dcompiler dwmapi imm32 gdi32 user32)
set_target_properties(hvh_menu PROPERTIES PREFIX "")
add_executable(hvh_loader src/loader.cpp)
target_include_directories(hvh_loader PRIVATE include)
add_executable(dx11_test_host WIN32 tests/dx11_host.cpp)
add_executable(feature_tests tests/features.cpp src/config.cpp src/game_data.cpp)
add_executable(aim_core_tests tests/aim_core.cpp)
target_include_directories(aim_core_tests PRIVATE src)
add_executable(aim_spread_tests tests/aim_spread.cpp)
target_include_directories(aim_spread_tests PRIVATE src)
add_executable(diagnostic_probe tools/diagnostic_probe.cpp src/game_data.cpp)
add_executable(skin_transaction_tests tests/skin_transactions.cpp src/config.cpp src/game_data.cpp
    ${MINHOOK}/src/buffer.c ${MINHOOK}/src/hook.c ${MINHOOK}/src/trampoline.c ${MINHOOK}/src/hde/hde64.c)
target_include_directories(skin_transaction_tests PRIVATE src ${MINHOOK}/include)
foreach(AIM_FIXTURE IN ITEMS aim_runtime aim_ballistics aim_actions aim_smoke aim_chance aim_hitboxes aim_performance esp)
    add_executable(${AIM_FIXTURE}_tests tests/${AIM_FIXTURE}.cpp src/config.cpp src/game_data.cpp src/server_rules.cpp src/esp.cpp
        ${MINHOOK}/src/buffer.c ${MINHOOK}/src/hook.c ${MINHOOK}/src/trampoline.c ${MINHOOK}/src/hde/hde64.c)
    target_include_directories(${AIM_FIXTURE}_tests PRIVATE src ${MINHOOK}/include)
    target_link_libraries(${AIM_FIXTURE}_tests PRIVATE user32)
    if(MINGW)
        target_link_options(${AIM_FIXTURE}_tests PRIVATE -static)
    endif()
endforeach()
foreach(RULE_FIXTURE IN ITEMS server_rules server_rule_filters)
    add_executable(${RULE_FIXTURE}_tests tests/${RULE_FIXTURE}.cpp src/config.cpp src/game_data.cpp
        ${MINHOOK}/src/buffer.c ${MINHOOK}/src/hook.c ${MINHOOK}/src/trampoline.c ${MINHOOK}/src/hde/hde64.c)
    target_include_directories(${RULE_FIXTURE}_tests PRIVATE src ${MINHOOK}/include)
    target_link_libraries(${RULE_FIXTURE}_tests PRIVATE user32)
    if(MINGW)
        target_link_options(${RULE_FIXTURE}_tests PRIVATE -static)
    endif()
endforeach()
target_include_directories(diagnostic_probe PRIVATE src)
target_include_directories(feature_tests PRIVATE src)
target_include_directories(dx11_test_host PRIVATE include src)
target_link_libraries(dx11_test_host PRIVATE d3d11 dxgi d3dcompiler user32)
if(MINGW)
    target_link_options(hvh_menu PRIVATE -static)
    target_link_options(hvh_loader PRIVATE -static -municode)
    target_link_options(dx11_test_host PRIVATE -static -municode)
    target_link_options(feature_tests PRIVATE -static)
    target_link_options(aim_core_tests PRIVATE -static)
    target_link_options(aim_spread_tests PRIVATE -static)
    target_link_options(diagnostic_probe PRIVATE -static)
    target_link_options(skin_transaction_tests PRIVATE -static)
endif()
